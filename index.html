Integration Code

Drop into any modern web framework in seconds

// 1. Initialize Neural Models
await Promise.all([
  faceapi.nets.tinyFaceDetector.loadFromUri('/models'),
  faceapi.nets.faceLandmark68Net.loadFromUri('/models'),
  faceapi.nets.faceRecognitionNet.loadFromUri('/models'),
]);

// 2. Detect Face & Extract 128-D Biometric Vector
const detection = await faceapi
  .detectSingleFace(videoElement, new faceapi.TinyFaceDetectorOptions())
  .withFaceLandmarks()
  .withFaceDescriptor();

// 3. Match against local enrolled faces
const matcher = new faceapi.FaceMatcher(enrolledDescriptors, 0.45);
const bestMatch = matcher.findBestMatch(detection.descriptor);
console.log(`Identified: ${bestMatch.label} (${Math.round((1 - bestMatch.distance) * 100)}%)`);
import { useFaceRecognition } from './useFaceRecognition';

export function FaceScannerComponent() {
  const { videoRef, status, recognizedUser, qualityScore } = useFaceRecognition({
    modelsUri: '/models',
    minConfidence: 0.70,
    onFaceMatch: (user) => console.log('Welcome back', user.name),
  });

  return (
    <div className="scanner-card">
      <video ref={videoRef} autoPlay muted playsInline />
      <p>Status: {status}</p>
      <p>User: {recognizedUser?.label || 'Searching...'}</p>
      <div className="bar" style={{ width: `${qualityScore}%` }} />
    </div>
  );
}
import { Component, ElementRef, ViewChild, OnInit } from '@angular/core';
import { FaceEngineService } from './face-engine.service';

@Component({
  selector: 'app-face-scanner',
  template: `
    <video #video autoplay muted playsinline></video>
    <p>Match: {{ currentMatch() }}</p>
  `
})
export class FaceScannerComponent implements OnInit {
  @ViewChild('video') videoEl!: ElementRef<HTMLVideoElement>;
  currentMatch = signal<string>('Searching');

  constructor(private faceEngine: FaceEngineService) {}

  async ngOnInit() {
    await this.faceEngine.loadModels('/models');
    await this.faceEngine.startCamera(this.videoEl.nativeElement);
  }
}
